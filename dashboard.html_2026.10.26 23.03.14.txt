        // Имитация данных сессии из БД для демонстрации интерфейса

        let isSealed = false;
        let currentStage = 'Neuromechanics'; 
        
        const mockRagRecords = [
            "[RAG_FIXED]: Обнаружена высокая склонность к рационализации этических дилемм.",
            "[RAG_FIXED]: Зафиксирован экзистенциальный страх ментального увядания.",

        // 1. КОНТУР АВТОРИЗАЦИИ ЛК (Квантово-защищенный ПИН-код)
        function unlockDashboard() {
            const passwordField = document.getElementById('gatePasswordField');
            const errorText = document.getElementById('gateErrorText');
            const pin = passwordField.value.trim();

            if (pin === "000000") {
                document.getElementById('auth-gate').style.opacity = '0';
                setTimeout(() => {
                    document.getElementById('auth-gate').style.display = 'none';
                    document.getElementById('dashboard-main').style.display = 'flex';
                    renderDashboardData();
                }, 500);
            } else {
                errorText.style.opacity = '1';
                passwordField.value = "";
            }
        }

        // 2. ОТРИСОВКА ИНТЕРФЕЙСА НА ОСНОВЕ ДАННЫХ СЕРВЕРА
        function renderDashboardData() {
            document.getElementById('userBalanceText').innerText = qcuBalance;
            document.getElementById('coreProgressBarFill').style.width = progress + "%";
            
            const statusDisplay = document.getElementById('coreStatusDisplay');
            const stageLabel = document.getElementById('coreStageLabel');
            const resumeBox = document.getElementById('resumeActionBox');
            const vaultBtn = document.getElementById('downloadVaultBtn');
            const vaultHint = document.getElementById('vaultHintText');
            const assistantBtn = document.getElementById('startAssistantBtn');

            const stageNames = {
                'Vectors': 'APXe_Vectors (0-30%)',
                'Neuromechanics': 'APXe_Neuromechanics (30-60%)',
                'Rems': 'APXe_Rems (60-90%)',
                'Mirror_Test': 'APXe_Mirror_Test (90-100%)',
                'Super_Calibration': 'СВЕРХ-КРИСТАЛЛИЗАЦИЯ (100%+)'
            };
            stageLabel.innerText = stageNames[currentStage] || 'КАЛИБРОВКА';

            if (progress >= 100.0 || isSealed) {
                statusDisplay.innerText = `ЯДРО ЗАПЕЧАТАНО [ПРИОРИТЕТНЫЙ СЛЕД: ${progress}%]`;
                resumeBox.innerHTML = `<button class="btn-action" onclick="startQuantumConduitLoad()">ДОКАЛИБРОВКА ЯДРА (100%+)</button>`;
                
                vaultBtn.disabled = false;
                vaultHint.innerText = "Контур верифицирован. Квантовый архив личности сгенерирован и заблокирован алгоритмом APXe_Safe_256 (AES-256 Постквантовая защита).";
                assistantBtn.disabled = false;
            } else {
                statusDisplay.innerText = `СИНХРОНИЗАЦИЯ: ${progress}% [КОНТУР ОТКРЫТ]`;
                resumeBox.innerHTML = `<button class="btn-action" onclick="startQuantumConduitLoad()">ПРОДОЛЖИТЬ СЕССИЮ</button>`;
                
                vaultBtn.disabled = true;
                assistantBtn.disabled = true;
            }

            const logsContainer = document.getElementById('ragLogsContainer');
            logsContainer.innerHTML = "";
            mockRagRecords.forEach(record => {
                const item = document.createElement('div');
                item.className = 'rag-item';
                item.innerText = record;
                logsContainer.appendChild(item);
            });
        }

        // 3. ЭЛИТНЫЙ КВАНТОВЫЙ ПРЕЛОАДЕР (Устранение черного экрана при переходе к TWA)
        function startQuantumConduitLoad() {
            // Скрываем дашборд, выводим оверлей прелоадера
            document.getElementById('dashboard-main').style.display = 'none';
            const preloader = document.getElementById('preloader-gate');
            preloader.style.display = 'flex';

            const stageText = document.getElementById('loaderStageText');
            const logsBox = document.getElementById('loaderLogsBox');

            // Массив бегущих логов для удержания внимания
            const preloaderLogs = [
                "[SYS]: Establishing E2EE quantum handshake...",
                "[SYS]: Direct connection to GPU cluster accepted.",
                "[SYS]: Initializing neuro-network weights...",
                "[RAG]: Loading personal knowledge graph database...",
                "[STT]: Synchronizing vocal frequency analyzers...",
                "[SYS]: Allocating server-side computing pool...",
                "[SYS]: Rendering 3D-core coordinate grid...",
                "[STATUS]: Identity matrix compilation - 100% OK."
            ];

            // Массив меняющихся заголовков по мере загрузки
            const stages = [
                "// ИНИЦИАЛИЗАЦИЯ ЗАЩИЩЕННОГО КОНТУРА //",
                "// ПОДКЛЮЧЕНИЕ К СУПЕРКОМПЬЮТЕРУ //",
                "// РАЗВЕРТЫВАНИЕ НЕЙРОМАТРИЦЫ //",
                "// СИНХРОНИЗАЦИЯ КВАНТОВОГО ШЛЮЗА //"
            ];

            let logIndex = 0;
            let stageIndex = 0;

            // Интервал вывода бегущих машинных логов
            const logInterval = setInterval(() => {
                if (logIndex < preloaderLogs.length) {
                    const newLog = document.createElement('div');
                    newLog.innerText = preloaderLogs[logIndex];
                    logsBox.appendChild(newLog);
                    logsBox.scrollTop = logsBox.scrollHeight;
                    logIndex++;
                }
            }, 6000); // Распределяем логи на минуту ожидания

            // Интервал смены технологических заголовков
            const stageInterval = setInterval(() => {
                if (stageIndex < stages.length) {
                    stageText.innerText = stages[stageIndex];
                    stageIndex++;
                }
            }, 12000);

            // Финальный триггер редиректа ровно через 55-60 секунд ожидания
            setTimeout(() => {
                clearInterval(logInterval);
                clearInterval(stageInterval);
                window.location.href = "./calibration.html";
            }, 55000);
        }

        // 4. БОКС КВАНТОВОГО СЕЙФА (Извлечение архива с паролем)
        function extractQuantumArchive() {
            if (qcuBalance < 10000) {
                alert("ОТКАЗ СЕТЕВОГО ШЛЮЗА: Недостаточно средств для декомпиляции весов модели. Требуется 10 000 QCU.");
                return;
            }

            const archivePassword = prompt("ВНИМАНИЕ: Запуск постквантового шифрования APXe_Safe_256.\n\nВведите секретный ключ для шифрования архива личности (минимум 8 символов).\n\nВНИМАНИЕ: ИИ зашьет этот ключ в структуру AES-256. Без него архив невозможно будет распаковать ни на одном компьютере мира:");
            
            if (!archivePassword || archivePassword.length < 8) {
                alert("ОШИБКА БЕЗОПАСНОСТИ: Длина ключа не соответствует протоколу постквантовой защиты.");
                return;
            }

            qcuBalance -= 10000;
            document.getElementById('userBalanceText').innerText = qcuBalance;

            alert("Инициализация глубокого рендеринга... Веса модели упакованы в контейнер .GGUF, защищены вашим персональным ключом и сжаты криптографическим методом AES-256.\n\nЗапуск скачивания квантового архива.");
            window.open("https://apxe.io", "_blank");
            renderDashboardData();
        }
